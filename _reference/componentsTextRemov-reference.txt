// components/TextRemoverHero.tsx
import React, { useCallback, useEffect, useRef, useState } from "react";

type Status = "idle" | "uploaded" | "processing" | "completed";

interface HighlightBox {
  id: number;
  x: number; // percentage
  y: number; // percentage
  w: number; // percentage
  h: number; // percentage
}

const DEMO_HIGHLIGHTS: HighlightBox[] = [
  { id: 1, x: 18, y: 22, w: 28, h: 9 },
  { id: 2, x: 52, y: 38, w: 32, h: 11 },
  { id: 3, x: 24, y: 58, w: 46, h: 10 },
  { id: 4, x: 14, y: 78, w: 20, h: 8 },
];

export default function TextRemoverHero() {
  const [status, setStatus] = useState<Status>("idle");
  const [originalUrl, setOriginalUrl] = useState<string>("");
  const [resultUrl, setResultUrl] = useState<string>("");
  const [sliderPct, setSliderPct] = useState<number>(50);
  const [isDragging, setIsDragging] = useState(false);
  const [fileName, setFileName] = useState<string>("");

  const containerRef = useRef<HTMLDivElement>(null);
  const inputRef = useRef<HTMLInputElement>(null);

  // Handle file selection
  const handleFile = useCallback((file: File) => {
    if (!file.type.startsWith("image/")) return;
    const url = URL.createObjectURL(file);
    setFileName(file.name);
    setOriginalUrl(url);
    setResultUrl("");
    setSliderPct(50);
    setStatus("uploaded");

    // Simulate AI detection + processing
    setTimeout(() => setStatus("processing"), 600);
    setTimeout(() => {
      setResultUrl(url); // In production, replace with API result
      setStatus("completed");
    }, 2800);
  }, []);

  const onDrop = useCallback(
    (e: React.DragEvent) => {
      e.preventDefault();
      const file = e.dataTransfer.files?.[0];
      if (file) handleFile(file);
    },
    [handleFile]
  );

  const onDragOver = useCallback((e: React.DragEvent) => {
    e.preventDefault();
  }, []);

  const onInputChange = useCallback(
    (e: React.ChangeEvent<HTMLInputElement>) => {
      const file = e.target.files?.[0];
      if (file) handleFile(file);
    },
    [handleFile]
  );

  // Slider drag logic
  const handleMove = useCallback(
    (clientX: number) => {
      if (!containerRef.current) return;
      const rect = containerRef.current.getBoundingClientRect();
      const pct = Math.min(Math.max((clientX - rect.left) / rect.width, 0), 1);
      setSliderPct(pct * 100);
    },
    []
  );

  const onMouseDown = useCallback(() => setIsDragging(true), []);
  const onMouseUp = useCallback(() => setIsDragging(false), []);
  const onMouseMove = useCallback(
    (e: React.MouseEvent) => {
      if (isDragging) handleMove(e.clientX);
    },
    [isDragging, handleMove]
  );
  const onTouchMove = useCallback(
    (e: React.TouchEvent) => {
      handleMove(e.touches[0].clientX);
    },
    [handleMove]
  );

  useEffect(() => {
    if (!isDragging) return;
    window.addEventListener("mouseup", onMouseUp);
    window.addEventListener("touchend", onMouseUp);
    return () => {
      window.removeEventListener("mouseup", onMouseUp);
      window.removeEventListener("touchend", onMouseUp);
    };
  }, [isDragging, onMouseUp]);

  const reset = useCallback(() => {
    setStatus("idle");
    setOriginalUrl("");
    setResultUrl("");
    setSliderPct(50);
    setFileName("");
    if (inputRef.current) inputRef.current.value = "";
  }, []);

  const downloadResult = useCallback(() => {
    if (!resultUrl) return;
    const a = document.createElement("a");
    a.href = resultUrl;
    a.download = fileName ? `clean-${fileName}` : "cleaned-image.png";
    a.click();
  }, [resultUrl, fileName]);

  return (
    <section className="min-h-screen w-full bg-[#FAFBFC] flex flex-col items-center justify-center px-4 py-6 md:py-10">
      {/* Header */}
      <div className="text-center mb-6 md:mb-8">
        <h1 className="text-3xl md:text-5xl font-bold text-slate-900 tracking-tight mb-3">
          Remove text from image{" "}
          <span className="text-[#6C5CE7]">in seconds</span>
        </h1>
        <p className="text-slate-500 text-base md:text-lg max-w-xl mx-auto">
          Upload a photo and let AI detect & erase unwanted text automatically.
        </p>
      </div>

      {/* Tool area: at least 70% viewport */}
      <div className="w-full max-w-6xl min-h-[70vh] bg-white rounded-2xl shadow-[0_8px_40px_-12px_rgba(108,92,231,0.18)] border border-slate-100 overflow-hidden flex flex-col">
        {/* Top action bar */}
        <div className="h-14 border-b border-slate-100 flex items-center justify-between px-5 md:px-7">
          <div className="flex items-center gap-2">
            <div className="w-3 h-3 rounded-full bg-[#6C5CE7]" />
            <div className="w-3 h-3 rounded-full bg-[#00B894]" />
            <div className="w-3 h-3 rounded-full bg-slate-200" />
          </div>
          <span className="text-sm font-medium text-slate-400">
            AI Text Remover
          </span>
        </div>

        {/* Main workspace */}
        <div className="flex-1 p-4 md:p-7 flex flex-col md:flex-row gap-5 md:gap-7">
          {status === "idle" ? (
            <div className="flex-1 flex items-center justify-center">
              <button
                onClick={() => inputRef.current?.click()}
                onDrop={onDrop}
                onDragOver={onDragOver}
                className="group w-full max-w-xl h-80 md:h-[420px] rounded-xl border-2 border-dashed border-slate-300 bg-gradient-to-b from-[#FAFBFC] to-white hover:border-[#6C5CE7] hover:shadow-[0_12px_40px_-12px_rgba(108,92,231,0.22)] transition-all duration-300 flex flex-col items-center justify-center gap-4"
              >
                <div className="w-16 h-16 rounded-2xl bg-[#6C5CE7]/10 flex items-center justify-center group-hover:scale-110 transition-transform duration-300">
                  <svg
                    className="w-8 h-8 text-[#6C5CE7]"
                    fill="none"
                    viewBox="0 0 24 24"
                    stroke="currentColor"
                    strokeWidth={1.8}
                  >
                    <path
                      strokeLinecap="round"
                      strokeLinejoin="round"
                      d="M3 16.5v2.25A2.25 2.25 0 005.25 21h13.5A2.25 2.25 0 0021 18.75V16.5m-13.5-9L12 3m0 0l4.5 4.5M12 3v13.5"
                    />
                  </svg>
                </div>
                <div className="text-center">
                  <p className="text-slate-700 font-medium text-lg">
                    Drop your image here or click to upload
                  </p>
                  <p className="text-slate-400 text-sm mt-1">
                    Supports JPG, PNG, WebP up to 10MB
                  </p>
                </div>
              </button>
              <input
                ref={inputRef}
                type="file"
                accept="image/*"
                className="hidden"
                onChange={onInputChange}
              />
            </div>
          ) : (
            <>
              {/* Left panel */}
              <div className="flex-1 flex flex-col min-h-[280px] md:min-h-0">
                <div className="flex items-center justify-between mb-3">
                  <span className="text-xs font-semibold uppercase tracking-wide text-slate-400">
                    Original
                  </span>
                  {status === "uploaded" && (
                    <span className="text-xs font-medium text-[#6C5CE7] bg-[#6C5CE7]/10 px-2.5 py-1 rounded-full animate-pulse">
                      Detecting text...
                    </span>
                  )}
                </div>

                <div className="flex-1 relative rounded-xl overflow-hidden bg-slate-50 border border-slate-100">
                  <img
                    src={originalUrl}
                    alt="Original"
                    className="w-full h-full object-contain"
                  />

                  {/* AI detected text highlights */}
                  {status !== "idle" && status !== "processing" && (
                    <div className="absolute inset-0 pointer-events-none">
                      {DEMO_HIGHLIGHTS.map((box) => (
                        <div
                          key={box.id}
                          className="absolute border-2 border-[#00B894] bg-[#00B894]/15 rounded-[4px] animate-pulse"
                          style={{
                            left: `${box.x}%`,
                            top: `${box.y}%`,
                            width: `${box.w}%`,
                            height: `${box.h}%`,
                          }}
                        />
                      ))}
                    </div>
                  )}

                  {status === "uploaded" && (
                    <div className="absolute inset-0 bg-[#6C5CE7]/5 backdrop-blur-[1px] flex items-center justify-center">
                      <div className="bg-white/90 px-4 py-2.5 rounded-full shadow-sm flex items-center gap-2">
                        <div className="w-2 h-2 rounded-full bg-[#6C5CE7] animate-bounce" />
                        <span className="text-sm font-medium text-slate-700">
                          Scanning for text regions
                        </span>
                      </div>
                    </div>
                  )}
                </div>
              </div>

              {/* Right panel / result */}
              <div className="flex-1 flex flex-col min-h-[280px] md:min-h-0">
                <div className="flex items-center justify-between mb-3">
                  <span className="text-xs font-semibold uppercase tracking-wide text-slate-400">
                    Result
                  </span>
                  {status === "completed" && (
                    <span className="text-xs font-medium text-[#00B894] bg-[#00B894]/10 px-2.5 py-1 rounded-full">
                      Done
                    </span>
                  )}
                </div>

                <div
                  ref={containerRef}
                  onMouseMove={onMouseMove}
                  onTouchMove={onTouchMove}
                  className="flex-1 relative rounded-xl overflow-hidden bg-slate-50 border border-slate-100 select-none"
                >
                  {status === "processing" ? (
                    <div className="w-full h-full flex flex-col items-center justify-center gap-4">
                      <div className="w-full max-w-xs space-y-3">
                        <div className="h-32 rounded-lg bg-slate-100 overflow-hidden relative">
                          <div className="absolute inset-0 bg-gradient-to-r from-transparent via-white/60 to-transparent animate-[shimmer_1.5s_infinite] -translate-x-full" />
                        </div>
                        <div className="h-4 rounded bg-slate-100 w-3/4 overflow-hidden relative">
                          <div className="absolute inset-0 bg-gradient-to-r from-transparent via-white/60 to-transparent animate-[shimmer_1.5s_infinite] -translate-x-full" />
                        </div>
                        <div className="h-4 rounded bg-slate-100 w-1/2 overflow-hidden relative">
                          <div className="absolute inset-0 bg-gradient-to-r from-transparent via-white/60 to-transparent animate-[shimmer_1.5s_infinite] -translate-x-full" />
                        </div>
                      </div>
                      <p className="text-sm font-medium text-slate-500 animate-pulse">
                        Removing text...
                      </p>
                    </div>
                  ) : status === "completed" ? (
                    <>
                      {/* Before/After comparison */}
                      <img
                        src={originalUrl}
                        alt="Before"
                        className="absolute inset-0 w-full h-full object-contain"
                      />
                      <div
                        className="absolute inset-0 overflow-hidden"
                        style={{ clipPath: `inset(0 ${100 - sliderPct}% 0 0)` }}
                      >
                        <img
                          src={resultUrl}
                          alt="After"
                          className="absolute inset-0 w-full h-full object-contain max-w-none"
                        />
                      </div>

                      {/* Slider divider */}
                      <div
                        className="absolute top-0 bottom-0 w-1 bg-white shadow-[0_0_12px_rgba(0,0,0,0.15)] cursor-ew-resize z-10"
                        style={{ left: `${sliderPct}%` }}
                        onMouseDown={onMouseDown}
                        onTouchStart={onMouseDown}
                      >
                        <div className="absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2 w-9 h-9 rounded-full bg-white shadow-lg border border-slate-100 flex items-center justify-center">
                          <svg
                            className="w-4 h-4 text-[#6C5CE7]"
                            fill="none"
                            viewBox="0 0 24 24"
                            stroke="currentColor"
                            strokeWidth={2}
                          >
                            <path
                              strokeLinecap="round"
                              strokeLinejoin="round"
                              d="M8 9l4-4 4 4m0 6l-4 4-4-4"
                            />
                          </svg>
                        </div>
                      </div>

                      {/* Labels */}
                      <span className="absolute top-3 left-3 text-xs font-semibold text-white bg-black/40 px-2 py-1 rounded backdrop-blur-sm">
                        Before
                      </span>
                      <span className="absolute top-3 right-3 text-xs font-semibold text-white bg-[#00B894]/80 px-2 py-1 rounded backdrop-blur-sm">
                        After
                      </span>
                    </>
                  ) : null}
                </div>
              </div>
            </>
          )}
        </div>

        {/* Bottom actions */}
        {status !== "idle" && (
          <div className="px-4 md:px-7 pb-6 md:pb-7 pt-2 flex flex-col sm:flex-row items-center justify-center gap-3">
            {status === "completed" && (
              <button
                onClick={downloadResult}
                className="w-full sm:w-auto px-7 py-3 rounded-xl bg-[#6C5CE7] hover:bg-[#5a4ed0] text-white font-semibold shadow-[0_8px_24px_-8px_rgba(108,92,231,0.5)] transition-all duration-200 flex items-center justify-center gap-2"
              >
                <svg
                  className="w-5 h-5"
                  fill="none"
                  viewBox="0 0 24 24"
                  stroke="currentColor"
                  strokeWidth={2}
                >
                  <path
                    strokeLinecap="round"
                    strokeLinejoin="round"
                    d="M3 16.5v2.25A2.25 2.25 0 005.25 21h13.5A2.25 2.25 0 0021 18.75V16.5M12 12.75V3m0 0l-3.75 3.75M12 3l3.75 3.75"
                  />
                </svg>
                Download
              </button>
            )}
            <button
              onClick={reset}
              className="text-sm font-medium text-slate-500 hover:text-[#6C5CE7] transition-colors underline-offset-4 hover:underline"
            >
              Try another image
            </button>
          </div>
        )}
      </div>

      {/* Tailwind keyframes for shimmer */}
      <style>{`
        @keyframes shimmer {
          100% {
            transform: translateX(100%);
          }
        }
      `}</style>
    </section>
  );
}
